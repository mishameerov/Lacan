cmake_minimum_required(VERSION 3.22)
project(lacan_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/whisper.cpp")
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/prepare-asr.sh before building")
endif()
add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/whisper")
add_library(lacan-whisper SHARED whisper-jni.cpp)
target_link_libraries(lacan-whisper PRIVATE whisper android log)
target_link_options(lacan-whisper PRIVATE "-Wl,-z,max-page-size=16384")
