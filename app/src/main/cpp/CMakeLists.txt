cmake_minimum_required(VERSION 3.22)
project(lacan_asr LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS lacan_compile=2 lacan_link=1)
set(CMAKE_JOB_POOL_COMPILE lacan_compile)
set(CMAKE_JOB_POOL_LINK lacan_link)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/whisper.cpp")
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/prepare-asr.sh before building")
endif()
add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/whisper")
add_library(lacan-whisper SHARED whisper-jni.cpp)
target_link_libraries(lacan-whisper PRIVATE whisper android log)
target_link_options(lacan-whisper PRIVATE "-Wl,-z,max-page-size=16384")

# GigaAM has a separate native library and a GPU-only MNN session. Preparation
# pins the source and installs the bounded execution-placement diagnostic hook.
set(MNN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/mnn/MNN-3.6.1")
set(KNF_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/knf/kaldi-native-fbank-1.22.3")
set(KISSFFT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.tools/kissfft/kissfft-febd4caeed32e33ad8b2e0bb5ea77542c40f18ec")
if(NOT EXISTS "${MNN_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${KNF_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${KISSFFT_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/prepare-giga-mnn.sh before building")
endif()
file(READ "${MNN_SOURCE}/source/core/Pipeline.cpp" MNN_PIPELINE_SOURCE)
if(NOT MNN_PIPELINE_SOURCE MATCHES "lacan_mnn_record_execution")
    message(FATAL_ERROR "MNN placement hook missing: run tools/prepare-giga-mnn.sh")
endif()
file(READ "${MNN_SOURCE}/source/backend/opencl/execution/buffer/ConvBufExecution.cpp" MNN_CONV_SOURCE)
if(NOT MNN_CONV_SOURCE MATCHES "lacan_clear_stale_conv_aux_kernels")
    message(FATAL_ERROR "MNN convolution resize patch missing: run tools/prepare-giga-mnn.sh")
endif()
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL ON CACHE BOOL "" FORCE)
set(MNN_ARM82 OFF CACHE BOOL "" FORCE)
set(MNN_INTERNAL OFF CACHE BOOL "" FORCE)
set(MNN_USE_LOGCAT ON CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
# These optional CPU kernels are unused by GPU inference; no build-time download.
set(MNN_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(MNN_SME2 OFF CACHE BOOL "" FORCE)
add_subdirectory("${MNN_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mnn")
set(KALDI_NATIVE_FBANK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(KALDI_NATIVE_FBANK_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(KALDI_NATIVE_FBANK_ENABLE_CHECK OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_KISSFFT "${KISSFFT_SOURCE}" CACHE PATH "" FORCE)
add_subdirectory("${KNF_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/knf")
add_library(lacan-giga-mnn SHARED gigaam-mnn-jni.cpp)
target_include_directories(lacan-giga-mnn PRIVATE "${MNN_SOURCE}/include" "${KNF_SOURCE}")
target_link_libraries(lacan-giga-mnn PRIVATE MNN kaldi-native-fbank-core android log)
target_link_options(lacan-giga-mnn PRIVATE "-Wl,-z,max-page-size=16384")
