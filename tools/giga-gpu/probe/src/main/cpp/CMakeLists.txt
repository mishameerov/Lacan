cmake_minimum_required(VERSION 3.22.1)
project(GigaGpuProbe LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(MNN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../.tools/mnn/MNN-3.6.1")
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_EXPRESS OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL ON CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_USE_LOGCAT ON CACHE BOOL "" FORCE)
set(MNN_ARM82 OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
add_subdirectory("${MNN_ROOT}" "${CMAKE_BINARY_DIR}/mnn")
add_library(gigaprobe SHARED probe.cpp)
target_include_directories(gigaprobe PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/source")
target_link_libraries(gigaprobe PRIVATE MNN android log)
target_link_options(gigaprobe PRIVATE "-Wl,-z,max-page-size=16384")
